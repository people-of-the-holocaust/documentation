\section{Acceptance Criteria}

\begin{longtable}{|p{0.12\textwidth}|p{0.78\textwidth}|}
	\hline
	\textbf{ID} & \textbf{Acceptance Criterion}                                                             \\
	\hline
	AC-1        & The homepage loads and shows the project purpose and description.                         \\
	\hline
	AC-2        & The People page loads person records from the database.                                   \\
	\hline
	AC-3        & Searching on the People page filters records based on entered text.                       \\
	\hline
	AC-4        & Selecting a person displays that person's details and relationships.                      \\
	\hline
	AC-5        & The Places page loads place records from the database.                                    \\
	\hline
	AC-6        & Searching on the Places page filters records based on entered text.                       \\
	\hline
	AC-7        & Selecting a place shows that place's details and related activities.                      \\
	\hline
	AC-8        & The map loads location records and shows markers for records with valid coordinates.      \\
	\hline
	AC-9        & Marker popups accurately show place information.                                          \\
	\hline
	AC-10       & The frontend build command completes successfully.                                        \\
	\hline
	AC-11       & The interpreter tests complete successfully.                                              \\
	\hline
	AC-12       & The interpreter logs failed processing attempts without stopping the full processing run. \\
	\hline
\end{longtable}
