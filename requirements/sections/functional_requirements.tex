\section{Functional Requirements}

\begin{longtable}{|p{0.12\textwidth}|p{0.68\textwidth}|p{0.12\textwidth}|}
	\hline
	\textbf{ID} & \textbf{Requirement}                                                                          & \textbf{Priority} \\
	\hline
	FR-1        & The system shall provide a public website for viewing \textit{People of the Holocaust} data.  & High              \\
	\hline
	FR-2        & The system shall not require users to create an account or log in to view public records.     & High              \\
	\hline
	FR-3        & The website shall provide a homepage that explains the background and purpose of the project. & High              \\
	\hline
	FR-4        & The website shall provide a People page that pull person records from the database.           & High              \\
	\hline
	FR-5        & The website shall allow users to search people by name.                                       & High              \\
	\hline
	FR-6        & The website shall always allow users to select a person and view related details.             & High              \\
	\hline
	FR-7        & The website shall provide a Places page that pulls place records from the database.           & High              \\
	\hline
	FR-8        & The website shall allow users to search places by name.                                       & High              \\
	\hline
	FR-9        & The website shall allow users to select a place and view related details.                     & High              \\
	\hline
	FR-10       & The website shall provide an interactive map of historical locations.                         & High              \\
	\hline
	FR-11       & The map shall display markers for places with valid latitude and longitude values.            & High              \\
	\hline
	FR-12       & The map shall display place information when a marker is selected.                            & High              \\
	\hline
	FR-13       & The interpreter shall clean raw person names into structured person records.                  & High              \\
	\hline
	FR-14       & The interpreter shall retrieve and clean article text from the \textit{ECG}.                  & High              \\
	\hline
	FR-15       & The interpreter shall identify sentences that contain known person names.                     & High              \\
	\hline
	FR-16       & The interpreter shall pull activity records from identified sentences when possible.          & High              \\
	\hline
	FR-17       & The interpreter shall associate activity records with people and places.                      & High              \\
	\hline
	FR-18       & The interpreter shall write processed records to project database.                            & High              \\
	\hline
	FR-19       & The interpreter shall log article processing logs for later review.                           & High              \\
	\hline
\end{longtable}
