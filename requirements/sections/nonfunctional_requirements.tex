\section{Non-Functional Requirements}

\begin{longtable}{|p{0.12\textwidth}|p{0.68\textwidth}|p{0.12\textwidth}|}
	\hline
	\textbf{ID} & \textbf{Requirement}                                                                                         & \textbf{Priority} \\
	\hline

	\hline
	NFR-2       & The website shall pull public data from the database through API requests or database client.                & High              \\

	\hline
	NFR-4       & The interpreter shall run Python 3.11 or newer.                                                              & High              \\
	\hline
	NFR-5       & The system shall preserve historical meaning of extracted material.                                          & High              \\
	\hline
	NFR-6       & The website shall remain usable when optional data fields are missing.                                       & High              \\
	\hline
	NFR-7       & Database credentials and service keys shall not be exposed in frontend code or committed to version control. & High              \\
	\hline
	NFR-8       & The system shall respect source permissions.                                                                 & High              \\
	\hline
	NFR-9       & The system shall be maintainable by future developers.                                                       & High              \\
	\hline
